### aloneguid on 2023-01-19

It should be easy to do - open file, read all columns, merge to the same columns from previous file and so on, then write out merged data. The data will have to be re-compressed due to how parquet format works.
However, if you need to merge, may I suggest the parallelism is not chosen right. Maybe it's worth to choose bigger chunks of work?

public static async Task<Stream> ConcatenateParquetFiles(IEnumerable<Stream> parquetFiles)
{
    if (parquetFiles == null || !parquetFiles.Any())
        throw new ArgumentNullException(nameof(parquetFiles), "At least one Parquet file must be provided.");

    var schema = (await ParquetReader.CreateAsync(parquetFiles.First())).Schema;

    // Count the total number of rows.
    long rowsCount = 0;
    foreach (var parquetFile in parquetFiles)
        using (var reader = await ParquetReader.CreateAsync(parquetFile))
            for (int i = 0; i < reader.RowGroupCount; i++)
                rowsCount += reader.OpenRowGroupReader(i).RowCount;

    // Allocate the storage for the concatenated columns.
    var columnsValues = new Array[schema.DataFields.Length];
    for (int i = 0; i < schema.DataFields.Length; ++i)
        columnsValues[i] = Array.CreateInstance(schema.DataFields[i].ClrType, rowsCount);

    // Copy the data from each column into the concatenated columns storage.
    var offset = 0;
    foreach (var parquetFile in parquetFiles)
        using (var reader = await ParquetReader.CreateAsync(parquetFile))
            for (int i = 0; i < reader.RowGroupCount; i++)
            {
                var columns = await reader.ReadEntireRowGroupAsync(i);

                for (int j = 0; j < columns.Length; j++)
                    Array.Copy(columns[j].Data, 0, columnsValues[j], offset, columns[j].Data.Length);

                offset += columns[0].Data.Length;
            }

    // Generate the concatenated Parquet file.
    var finalParquetFile = new MemoryStream();

    using (var writer = await ParquetWriter.CreateAsync(schema, finalParquetFile))
    using (var rg = writer.CreateRowGroup())
        for (int i = 0; i < columnsValues.Length; ++i)
            await rg.WriteColumnAsync(new DataColumn(schema.DataFields[i], columnsValues[i]));

    finalParquetFile.Seek(0, SeekOrigin.Begin);

    return finalParquetFile;
}
