        }

        [Fact]

            var schema = new ParquetSchema(
                new DataField("S1", typeof(string), isNullable: false));

            var ms = new MemoryStream();

                using ParquetWriter writer = ParquetWriter.CreateAsync(schema, ms).GetAwaiter().GetResult();
                using ParquetRowGroupWriter groupWriter = writer.CreateRowGroup();

                groupWriter.WriteColumnAsync(
                        new DataColumn(

                    .GetResult();

### ebgieseking on 2024-11-21

For those that need to deserialize a parquet file that was created with a non-nullable property into a class with a nullable property, or want to pass custom JsonSerializerSettings for other reasons, here's a solution using dynamic types and newtonsoft json to work around the ParquetSerializer logic:

```
public static async Task<List<T?>> DeserializeAsync<T>(Stream fileStream, JsonSerializerSettings serializerSettings = null!, CancellationToken cancellationToken = default) where T : class
    {
        serializerSettings ??= new JsonSerializerSettings();
        var result = await ParquetSerializer.DeserializeAsync(fileStream, cancellationToken: cancellationToken);

        return result.Data
            .Select(kvp =>
            {
                var expando = new ExpandoObject() as IDictionary<string, object>;
                foreach (var property in kvp)
                {
                    expando.Add(property.Key, property.Value);
                }
                return JsonConvert.DeserializeObject<T>(JsonConvert.SerializeObject(expando), serializerSettings);
            })
            .ToList();
    }
```
