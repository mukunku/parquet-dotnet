[Fact]
        public async Task Defect() 
        {
            var filename = "d:\\schema.parquet";

            var field = new DateTimeDataField("Date", DateTimeFormat.DateAndTime, true);
            var schema = new ParquetSchema(field);

            using(var fileStream = new FileStream(filename, FileMode.Create, FileAccess.Write, FileShare.None)) {

                using(var parquetWriter = await ParquetWriter.CreateAsync(schema, fileStream)) 
                {
                    using(var groupWriter = parquetWriter.CreateRowGroup()) {

                        var dates = new List<DateTime>() { DateTime.Now };
                        var dataColumn = new DataColumn(field, dates.ToArray());

                        groupWriter.WriteColumnAsync(dataColumn);
                    }
                }
            }


            using(var parquetReader = await ParquetReader.CreateAsync(filename)) 
            {
                parquetReader.Schema.Fields.ToString();

                Assert.Single(schema.Fields);
                Assert.Equal(schema.Fields.Count, parquetReader.Schema.Fields.Count);
                Assert.Equal(schema.Fields.First().GetType(), parquetReader.Schema.Fields.First().GetType());
            }
        }
