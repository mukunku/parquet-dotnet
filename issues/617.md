```csharp
public class LowLevelParquetReader
{
    public static async Task ReadAndGroupToJsonAsync(string directoryPath, string outputJsonPath)
    {
        var groupedRows = new List<Dictionary<string, object?>>();
        string[] parquetFiles = Directory.GetFiles(directoryPath, "*.parquet");


        foreach (var file in parquetFiles)
        {
            await using Stream fs = File.OpenRead(file);
            using ParquetReader reader = await ParquetReader.CreateAsync(fs);
            for (int i = 0; i < reader.RowGroupCount; i++)
            {
                using ParquetRowGroupReader rowGroupReader = reader.OpenRowGroupReader(i);
                var dataFields = reader.Schema.GetDataFields();
                var columns = new List<DataColumn>();
                foreach (DataField df in dataFields)
                {
                    var column = await rowGroupReader.ReadColumnAsync(df);
                    columns.Add(column);
                }

                var rowDict = new Dictionary<string, object?>();
                foreach (var col in columns)
                {
                    rowDict[col.Field.Name] = col.Data.GetValue(0)?.ToString();
                }

                groupedRows.Add(rowDict);
            }
        }

        var json = JsonSerializer.Serialize(groupedRows, new JsonSerializerOptions { WriteIndented = true });
        await File.WriteAllTextAsync(outputJsonPath, json, Encoding.UTF8);
    }
}
```
