### How to reproduce?

Generate parquet files with nested class vs readonly struct

using Parquet.Serialization;

namespace MCSRecorder
{
    readonly struct SConstituent
    {
        public readonly string Symbol;
        public readonly string Name;

        public SConstituent(string symbol, string name)
        {
            Symbol = symbol;
            Name = name;
        }
    }

    class Constituent
    {
        public readonly string Symbol;
        public readonly string Name;

        public Constituent(string symbol, string name)
        {
            Symbol = symbol;
            Name = name;
        }
    }

    class Order
    {
        public readonly DateTime Timestamp;
        public readonly int Id;
        public readonly double Price;
        public readonly Constituent Constituent;
        public readonly Constituent[] Constituents;
        public readonly SConstituent[] SConstituents;

        public Order(DateTime timestamp, int id, double price, Constituent constituent, Constituent[] constituents, SConstituent[] sConstituents)
        {
            Timestamp = timestamp;
            Id = id;
            Price = price;
            Constituent = constituent;
            Constituents = constituents;
            SConstituents = sConstituents;
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            var data = Enumerable.Range(0, 1_000).Select(i => new Order(
                timestamp: DateTime.UtcNow,
                id: i,
                price: i,
                constituent: new Constituent(symbol: "APPL", name: "P"),
                constituents:
                [
                    new Constituent(symbol: "APPL", name: "P"),
                    new Constituent(symbol: "Micro", name: "M"),
                ],
                sConstituents:
                [
                    new SConstituent(symbol: "APPL", name: "P"),
                    new SConstituent(symbol: "Micro", name: "M"),
                    ])
            ).ToList();

            ParquetSerializer.SerializeAsync(data, "C:\\Users\\MB6291\\Downloads\\mcs.records").Wait();
        }
    }
}
