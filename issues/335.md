### atlemann on 2023-05-19

Polygot notebooks in VSCode doesn't play well with `using` expressions, so I just disposed the things I remembered inside each cell. It seems I forgot the row group reader. Disposing the row group writer seems to do the trick.

I got fooled by Parquet.Net and pyarrow being able to read the files, but other libs weren't. I slapped together a console app instead and that works as well.

I found the bug in my code which was the opposite. The row group writer went out of scope and was disopsed before all columns were written.

So I'm sorry for bothering you with this. My bad.

```csharp
using Parquet;
using Parquet.Data;
using Parquet.Schema;

using ParquetSharp;

using System.IO;
using System.Linq;

async Task Write(FileInfo destination)
{
    var fields = new DataField[] { new DataField<string>("Names"), new DataField<double>("Values") };
    var schema = new ParquetSchema(fields);

    using var stream = destination.OpenWrite();
    using var writer = await ParquetWriter.CreateAsync(schema, stream);
    using var rg = writer.CreateRowGroup();

    var names = new DataColumn(fields[0], new [] { "Name 1", "Name 2", "Name 3" });
    var values = new DataColumn(fields[1], new [] { 1.0, 2.0, 3.0 });

    await rg.WriteColumnAsync(names);
    await rg.WriteColumnAsync(values);
}

async Task<(string, double)[]> ReadWithParquetNet(FileInfo file)
{
    // Read with Parquet.Net
    using var stream = file.OpenRead();
    using var reader = await ParquetReader.CreateAsync(stream);
    using var rg = reader.OpenRowGroupReader(0);

    var fields = reader.Schema.DataFields;
    var ns = await rg.ReadColumnAsync(fields[0]);
    var vs = await rg.ReadColumnAsync(fields[1]);

    var zipped = ((string[]) ns.Data).Zip(((double[])vs.Data)).ToArray();
    return zipped;
}

(string, double)[] ReadWithParquetSharp(FileInfo file)
{
    // Read with ParquetSharp
    using var sharpReader = new ParquetFileReader(file.FullName);
    using var rg = sharpReader.RowGroup(0);

    var groupNumRows = checked((int) rg.MetaData.NumRows);
    var names = rg.Column(0).LogicalReader<string>().ReadAll(groupNumRows);
    var values = rg.Column(1).LogicalReader<double>().ReadAll(groupNumRows);

    var zipped = names.Zip(values).ToArray();
    return zipped;
}


var file = new FileInfo("/tmp/test.parquet");

await Write(file);

Console.WriteLine("Parquet.Net:");
var parquetNet = await ReadWithParquetNet(file);
foreach (var (name, value) in parquetNet)
    Console.WriteLine($"{name} - {value}");

Console.WriteLine("ParquetSharp:");
var parquetSharp = ReadWithParquetSharp(file);
foreach (var (name, value) in parquetSharp)
    Console.WriteLine($"{name} - {value}");
```
