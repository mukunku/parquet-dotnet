## Description

I created a new .Net 7 console project and wrote some test code against Parquet, and it seems like the file it creates is corrupted. Mind taking a look?

project file:
<Project Sdk="Microsoft.NET.Sdk">

    <PropertyGroup>
        <OutputType>Exe</OutputType>
        <TargetFramework>net7.0</TargetFramework>
        <ImplicitUsings>enable</ImplicitUsings>
        <Nullable>enable</Nullable>
    </PropertyGroup>

    <ItemGroup>
      <PackageReference Include="Parquet.Net" Version="4.4.4" />
    </ItemGroup>

</Project>


Program.cs:
```
// See https://aka.ms/new-console-template for more information

using Parquet;
using Parquet.Data;
using Parquet.Schema;
using Parquet.Thrift;

Console.WriteLine("Hello, World!");

var dir = new DirectoryInfo(Environment.CurrentDirectory).Parent.Parent.Parent;
Console.WriteLine(dir.FullName);


//create data columns with schema metadata and the data you need
var idField = new DataField<int>("id");

var cityField = new DataField<string>("city");


// create file schema
var schema = new ParquetSchema(cityField, idField);

var rnd = new Random();

int[] GenIds(int rows)
{
    var result = new int[rows];
    for (var i = 0; i < rows; i++)
    {
        result[i] = i + 2 * 1_000_000;
    }
    return result;
}


string[] GenCities(int rows)
{
    var result = new string[rows];
    for (var i = 0; i < rows; i++)
    {
        var vals = new[] { rnd.Next(65, 122), rnd.Next(65, 122), rnd.Next(65, 122), };
        result[i] = string.Join("", vals.Select(v => (char)v));
    }
    
    if (result.Any(s => s == null))
    {
        throw new Exception("I'm attempting to store nulls");
    }
    
    return result;
}

async Task CreateFile()
{
    var file = new FileInfo(Path.Combine(dir.FullName, "string.parquet"));
    if (file.Exists)
        file.Delete();
    
    await using var fileStream = File.OpenWrite(file.FullName);
    using var parquetWriter = await ParquetWriter.CreateAsync(schema, fileStream);
// create a new row group in the file
    using var groupWriter = parquetWriter.CreateRowGroup();
    await groupWriter.WriteColumnAsync(new DataColumn(cityField, GenCities(1_000_000).ToArray()));
    await groupWriter.WriteColumnAsync(new DataColumn(idField, GenIds(1_000_000)));
}


async Task ReadFile()
{
    var file = new FileInfo(Path.Combine(dir.FullName, "string.parquet"));
    
    await using var fileStream = File.OpenRead(file.FullName);
    using var parquetReader = await ParquetReader.CreateAsync(fileStream);
    for (var i = 0; i < parquetReader.RowGroupCount; i++)
    {
        using var grp = parquetReader.OpenRowGroupReader(i);
        var city = parquetReader.Schema.GetDataFields().Single(f => f.Name == "city");
        var col = await grp.ReadColumnAsync(city);
        var a = (string[]) col.Data;

        if (a.Any(s => s == null))
        {
            throw new Exception("I didn't store any nulls");
        }
        
    }
}

await CreateFile();


await ReadFile();

Console.WriteLine("Done!");
```

When I run this code, I get the thrown exception
`Unhandled exception. System.Exception: I didn't store any nulls`
